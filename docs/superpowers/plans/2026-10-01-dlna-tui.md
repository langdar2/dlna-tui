# DLNA TUI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Eine Terminal-Vollbild-App, die Musik auf einem DLNA-Medienserver durchsucht und auf einem DLNA-Renderer abspielt.

**Architecture:** `dlna/` kapselt UPnP (keine Textual-Abhängigkeit, pur testbar), `ui/` ist die Textual-TUI (kein SOAP). Austausch nur über dataclasses aus `dlna/models.py`. Die Lib `async-upnp-client` liefert fertige DLNA-Profile (`DmsDevice`/`DmrDevice`) und gebündeltes `didl_lite` — wir schreiben kein SOAP/DIDL-XML selbst, sondern dünne Adapter auf unsere Modelle.

**Tech Stack:** Python 3.13 (async), `async-upnp-client==0.46.2` (bereits installiert), `textual`, `pytest`.

```python
from __future__ import annotations

from dataclasses import dataclass, field
from didl_lite import didl_lite


@dataclass
class DeviceInfo:
    udn: str
    name: str
    location: str
    kind: str  # "server" | "renderer"


@dataclass
class Container:
    id: str
    title: str
    child_count: int | None = None


@dataclass
class Track:
    id: str
    title: str
    artist: str = ""
    album: str = ""
    duration: str = ""
    media_url: str = ""
    didl: str = ""  # DIDL-Lite-Fragment für den Renderer


def _first_res_uri(obj) -> str:
    for res in getattr(obj, "res", []) or []:
        if getattr(res, "uri", None):
            return res.uri
    return ""


def _first_res_duration(obj) -> str:
    for res in getattr(obj, "res", []) or []:
        if getattr(res, "duration", None):
            return res.duration
    return ""


def _to_model(obj):
    if isinstance(obj, didl_lite.Container):
        cc = obj.child_count
        return Container(
            id=obj.id,
            title=obj.title or "",
            child_count=int(cc) if cc not in (None, "") else None,
        )
    return Track(
        id=obj.id,
        title=obj.title or "",
        artist=getattr(obj, "artist", "") or "",
        album=getattr(obj, "album", "") or "",
        duration=_first_res_duration(obj),
        media_url=_first_res_uri(obj),
        didl=obj.to_xml_string().decode() if hasattr(obj, "to_xml_string") else "",
    )


def from_didl(didl_xml: str) -> list[Container | Track]:
    """DIDL-Lite-XML -> Liste unserer Modelle (Descriptors verworfen)."""
    out: list[Container | Track] = []
    for obj in didl_lite.from_xml_string(didl_xml, strict=False):
        if isinstance(obj, (didl_lite.Container, didl_lite.Item)):
            out.append(_to_model(obj))
    return out
```

```python
@dataclass
class Queue:
    tracks: list[Track] = field(default_factory=list)
    index: int = 0

    def current(self) -> Track | None:
        if 0 <= self.index < len(self.tracks):
            return self.tracks[self.index]
        return None

    def advance(self) -> Track | None:
        if self.index + 1 < len(self.tracks):
            self.index += 1
            return self.tracks[self.index]
        return None

    def prev(self) -> Track | None:
        if self.index > 0:
            self.index -= 1
            return self.tracks[self.index]
        return None

    def append(self, track: Track) -> None:
        self.tracks.append(track)
```

```python
from __future__ import annotations

import json
from pathlib import Path


def default_path() -> Path:
    return Path.home() / ".config" / "dlna-tui" / "store.json"


class Store:
    def __init__(self, path: Path | None = None):
        self.path = Path(path) if path else default_path()
        self.favorites: list[dict] = []
        self.last_server: str | None = None
        self.last_renderer: str | None = None
        self._load()

    def _load(self) -> None:
        try:
            data = json.loads(self.path.read_text())
        except (OSError, ValueError):
            return  # fehlt oder korrupt -> Defaults
        self.favorites = data.get("favorites", [])
        self.last_server = data.get("last_server")
        self.last_renderer = data.get("last_renderer")

    def save(self) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.path.write_text(json.dumps({
            "favorites": self.favorites,
            "last_server": self.last_server,
            "last_renderer": self.last_renderer,
        }, indent=2))
```

- [ ] **Step 1: Implementieren**

`dlna/discovery.py`:

```python
from __future__ import annotations

import asyncio

from async_upnp_client.aiohttp import AiohttpRequester
from async_upnp_client.client_factory import UpnpFactory
from async_upnp_client.profiles.dlna import DmrDevice, DmsDevice

from .models import DeviceInfo


async def _discover_kind(profile, kind: str, timeout: float) -> list[DeviceInfo]:
    found: dict[str, DeviceInfo] = {}
    for entry in await profile.async_discover():
        location = entry.get("location")
        udn = entry.get("usn", "").split("::")[0]
        if not location or udn in found:
            continue
        found[udn] = DeviceInfo(udn=udn, name=udn, location=location, kind=kind)
    return list(found.values())


async def discover(timeout: float = 3.0) -> tuple[list[DeviceInfo], list[DeviceInfo]]:
    """SSDP-Scan. Gibt (servers, renderers) zurück. Namen werden beim Verbinden ergänzt."""
    servers, renderers = await asyncio.gather(
        _discover_kind(DmsDevice, "server", timeout),
        _discover_kind(DmrDevice, "renderer", timeout),
    )
    return servers, renderers


async def build_device(info: DeviceInfo):
    """DeviceInfo -> konkretes UpnpDevice (für MusicServer/Renderer)."""
    requester = AiohttpRequester()
    factory = UpnpFactory(requester)
    return await factory.async_create_device(info.location)
```

```python
from __future__ import annotations

from .models import Container, Track, from_didl


class MusicServer:
    def __init__(self, profile):
        self._p = profile  # async_upnp_client DmsDevice

    async def browse(self, object_id: str = "0") -> list[Container | Track]:
        result = await self._p.async_browse_direct_children(object_id)
        return from_didl(result.result)

    async def search(self, query: str, container_id: str = "0") -> list[Track]:
        criteria = (
            f'upnp:class derivedfrom "object.item.audioItem" '
            f'and dc:title contains "{query}"'
        )
        try:
            result = await self._p.async_search_directory(container_id, criteria)
            return [o for o in from_didl(result.result) if isinstance(o, Track)]
        except Exception:
            # ponytail: Server ohne Search -> Browse + lokaler Filter.
            items = await self.browse(container_id)
            q = query.lower()
            return [o for o in items if isinstance(o, Track) and q in o.title.lower()]
```

```python
from __future__ import annotations

from dataclasses import dataclass
from typing import Callable

from .models import Track


@dataclass
class RendererState:
    transport_state: str = ""
    position: int = 0
    duration: int = 0
    volume: float = 0.0
    title: str = ""


class Renderer:
    def __init__(self, profile, on_error: Callable[[str], None] | None = None):
        self._p = profile  # async_upnp_client DmrDevice
        self._on_error = on_error or (lambda msg: None)

    async def _safe(self, coro) -> bool:
        try:
            await coro
            return True
        except Exception as exc:  # ponytail: UI darf nie an UPnP-Fehlern sterben
            self._on_error(str(exc))
            return False

    def state(self) -> RendererState:
        return RendererState(
            transport_state=self._p.transport_state or "",
            position=int(self._p.media_position or 0),
            duration=int(self._p.media_duration or 0),
            volume=float(self._p.volume_level or 0.0),
            title=self._p.media_title or "",
        )

    async def play(self, track: Track) -> bool:
        async def _go():
            await self._p.async_set_transport_uri(
                track.media_url, track.title, track.didl or None)
            await self._p.async_play()
        return await self._safe(_go())

    async def pause(self) -> bool:
        return await self._safe(self._p.async_pause())

    async def stop(self) -> bool:
        return await self._safe(self._p.async_stop())

    async def next(self) -> bool:
        return await self._safe(self._p.async_next())

    async def previous(self) -> bool:
        return await self._safe(self._p.async_previous())

    async def seek(self, seconds: int) -> bool:
        return await self._safe(self._p.async_seek_rel_time(seconds))

    async def set_volume(self, level: float) -> bool:
        level = max(0.0, min(1.0, level))
        return await self._safe(self._p.async_set_volume_level(level))

    async def subscribe(self, on_event: Callable[[], None]) -> bool:
        self._p.on_event = lambda svc, vars: on_event()
        return await self._safe(self._p.async_subscribe_services(auto_resubscribe=True))
```

Hinweis: `async_pause`, `async_seek_rel_time` etc. existieren auf `DmrDevice` (siehe Profil-API). `coro`-Objekte müssen innerhalb `_safe` awaited werden — die `play`-Methode baut deshalb eine innere Coroutine.

```python
from textual.reactive import reactive
from textual.widgets import Static

from dlna.renderer import RendererState


def _fmt(seconds: int) -> str:
    m, s = divmod(int(seconds), 60)
    return f"{m}:{s:02d}"


class NowPlaying(Static):
    st: reactive[RendererState | None] = reactive(None)

    def render(self) -> str:
        if not self.st or not self.st.title:
            return "— nichts spielt —"
        s = self.st
        return (f"▶ {s.title}   {_fmt(s.position)}/{_fmt(s.duration)}   "
                f"vol {int(s.volume * 100)}%   [{s.transport_state}]")
```

```python
        ("f", "favorite", "Fav"),
        ("r", "rescan", "Rescan"),
        ("q", "quit", "Quit"),
    ]

    def __init__(self, server: MusicServer, renderer: Renderer, store: Store):
        super().__init__()
        self.server = server
        self.renderer = renderer
        self.store = store
        self.queue = Queue()
        self._node_obj: dict = {}

    def compose(self) -> ComposeResult:
        yield Header()
        with Horizontal():
            yield Tree("Server", id="browser")
            yield ListView(id="queue")
        yield NowPlaying(id="now")
        yield Footer()

    async def on_mount(self) -> None:
        tree = self.query_one("#browser", Tree)
        await self._load_children(tree.root, "0")
        tree.root.expand()
        self.set_interval(1.0, self._refresh_state)
        await self.renderer.subscribe(self._refresh_state)

    async def _load_children(self, node, object_id: str) -> None:
        for obj in await self.server.browse(object_id):
            child = node.add(obj.title, allow_expand=isinstance(obj, Container))
            self._node_obj[child.id] = obj

    async def on_tree_node_expanded(self, event: Tree.NodeExpanded) -> None:
        obj = self._node_obj.get(event.node.id)
        if isinstance(obj, Container) and not event.node.children:
            await self._load_children(event.node, obj.id)

    async def on_tree_node_selected(self, event: Tree.NodeSelected) -> None:
        obj = self._node_obj.get(event.node.id)
        if isinstance(obj, Track):
            self.queue = Queue([obj])
            await self.renderer.play(obj)

    def _refresh_state(self) -> None:
        self.query_one("#now", NowPlaying).st = self.renderer.state()
        if self.renderer.state().transport_state == "STOPPED":
            nxt = self.queue.advance()
            if nxt:
                asyncio.create_task(self.renderer.play(nxt))

    async def action_toggle(self) -> None:
        st = self.renderer.state().transport_state
        await (self.renderer.pause() if st == "PLAYING" else self.renderer.play(self.queue.current())) if self.queue.current() else None

    async def action_next(self) -> None:
        nxt = self.queue.advance()
        if nxt:
            await self.renderer.play(nxt)

    async def action_prev(self) -> None:
        prv = self.queue.prev()
        if prv:
            await self.renderer.play(prv)

    async def action_enqueue(self) -> None:
        node = self.query_one("#browser", Tree).cursor_node
        obj = self._node_obj.get(node.id) if node else None
        if isinstance(obj, Track):
            self.queue.append(obj)
            self.query_one("#queue", ListView).append(ListItem(Label(obj.title)))

    async def action_vol_up(self) -> None:
        await self.renderer.set_volume(self.renderer.state().volume + 0.05)

    async def action_vol_down(self) -> None:
        await self.renderer.set_volume(self.renderer.state().volume - 0.05)

    async def action_seek_fwd(self) -> None:
        await self.renderer.seek(10)

    async def action_seek_back(self) -> None:
        await self.renderer.seek(-10)

    async def action_search(self) -> None:
        await self.push_screen(SearchScreen(self))

    async def action_favorite(self) -> None:
        node = self.query_one("#browser", Tree).cursor_node
        obj = self._node_obj.get(node.id) if node else None
        if isinstance(obj, Track):
            self.store.favorites.append({"id": obj.id, "title": obj.title,
                                         "media_url": obj.media_url, "didl": obj.didl})
            self.store.save()
            self.notify(f"Favorit: {obj.title}")

    async def action_rescan(self) -> None:
        self.notify("Rescan: Neustart der App nötig (d für Geräte).")

    async def action_quit(self) -> None:
        self.store.save()
        self.exit()
```

```python
from __future__ import annotations

import asyncio
import sys

from async_upnp_client.aiohttp import AiohttpRequester
from async_upnp_client.event_handler import UpnpEventHandler
from async_upnp_client.profiles.dlna import DmrDevice, DmsDevice

from dlna.discovery import build_device, discover
from dlna.renderer import Renderer
from dlna.server import MusicServer
from dlna.store import Store
from ui.app import DlnaApp


def _pick(devices, label, preferred_udn):
    if not devices:
        print(f"Kein {label} gefunden.")
        sys.exit(1)
    for d in devices:
        if d.udn == preferred_udn:
            return d
    if len(devices) == 1:
        return devices[0]
    for i, d in enumerate(devices):
        print(f"[{i}] {label}: {d.name} ({d.location})")
    return devices[int(input(f"{label} wählen: "))]


async def main() -> None:
    store = Store()
    print("Suche Geräte…")
    servers, renderers = await discover()
    srv_info = _pick(servers, "Server", store.last_server)
    rnd_info = _pick(renderers, "Renderer", store.last_renderer)

    srv_dev = await build_device(srv_info)
    rnd_dev = await build_device(rnd_info)

    requester = AiohttpRequester()
    event_handler = UpnpEventHandler("http://0.0.0.0:0", requester)

    server = MusicServer(DmsDevice(srv_dev, event_handler=event_handler))
    renderer = Renderer(DmrDevice(rnd_dev, event_handler=event_handler))

    store.last_server = srv_info.udn
    store.last_renderer = rnd_info.udn
    store.save()

    await DlnaApp(server, renderer, store).run_async()


if __name__ == "__main__":
    asyncio.run(main())
```
